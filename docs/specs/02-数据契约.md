## 2.1 数学类型 —— `src/math/vec3.ts`、`src/math/quat.ts`

```ts
// src/math/vec3.ts
export type V3 = [number, number, number];
export const add = (a: V3, b: V3): V3 => [a[0] + b[0], a[1] + b[1], a[2] + b[2]];
export const sub = (a: V3, b: V3): V3 => [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
export const mul = (a: V3, s: number): V3 => [a[0] * s, a[1] * s, a[2] * s];
export const dot = (a: V3, b: V3): number => a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
export const cross = (a: V3, b: V3): V3 => [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]];
export const length = (a: V3): number => Math.sqrt(dot(a, a));
export const norm = (a: V3): V3 => { const l = length(a); return l < 1e-9 ? [0, 0, 0] : mul(a, 1 / l); };
```

```ts
// src/math/quat.ts   四元数顺序 [w, x, y, z]
import { V3, add, mul, cross } from './vec3';
export type Q = [number, number, number, number];
export function qMul(a: Q, b: Q): Q;        // 与 sim_reference.q_mul 完全相同
export function qRot(q: Q, v: V3): V3;      // 与 q_rot 相同：t = 2*cross(u,v); v + w*t + cross(u,t)
export function qInvRot(q: Q, v: V3): V3;   // qRot([w,-x,-y,-z], v)
export function qNorm(q: Q): Q;
export function qFromYaw(yaw: number): Q;   // [cos(yaw/2), 0, sin(yaw/2), 0]
```

```ts
// src/math/scalar.ts
export const clamp = (x: number, lo: number, hi: number) => (x < lo ? lo : x > hi ? hi : x);
export const sign = (x: number) => (x > 0 ? 1 : x < 0 ? -1 : 0);
export const lerp = (a: number, b: number, t: number) => a + (b - a) * t;
export const smoothstep = (e0: number, e1: number, x: number) => { const t = clamp((x - e0) / (e1 - e0), 0, 1); return t * t * (3 - 2 * t); };
export const expDecay = (rate: number, dt: number) => 1 - Math.exp(-rate * dt); // 指数平滑系数
export const wrapAngle = (a: number) => ((a + Math.PI) % (2 * Math.PI) + 2 * Math.PI) % (2 * Math.PI) - Math.PI;
```

> 注意：Python 的 `%` 对负数取正余数，JS 的 `%` 保留符号。**移植 Python 里所有的 `%` 时一律写成 `((a % b) + b) % b`**。

export type Axle = 'front' | 'rear';

```ts
export type QualityLevel = 'auto' | 'low' | 'medium' | 'high';
export type CameraMode = 'chase' | 'chaseFar' | 'hood' | 'bumper';
export interface Settings {
  version: 1;

  camera: CameraMode;
  showGhost: boolean;
  volume: { master: number; engine: number; effects: number };   // 0..1

  version: 1, quality: 'auto',
  assists: { abs: true, tcs: true, esc: true, autoGear: true },
  steeringAssist: { keyboard: true, gamepad: false, touch: true },
  camera: 'chase', showGhost: true,

  paint: { hatch: 0, coupe: 0, rally: 0 },
};

export interface BestRecord { lapMs: number; sectorsMs: [number, number, number]; dateIso: string; assists: Assists; }
```

localStorage 的键（全部带版本号）：

| 键 | 值 |
|---|---|
| `apex.settings.v1` | `JSON.stringify(Settings)` |
| `apex.best.v1.<trackId>.<carId>` | `JSON.stringify(BestRecord)` |
| `apex.ghost.v1.<trackId>.<carId>` | 幽灵车数据：base64 编码的 Float32Array，格式见 07 §7.6 |

所有读写都要包在 `try/catch` 里。读取失败或 JSON 解析失败时返回默认值；写入失败时静默忽略，只 `console.warn` 一次。
